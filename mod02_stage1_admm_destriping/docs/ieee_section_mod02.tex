\subsection{Stage 1: ADMM High-Order Directional Destriping}
\label{subsec:stage1_admm}

\subsubsection{Constrained Optimization Formulation}
To disentangle the low-rank column-correlated stripe noise $\mathbf{S}$ from the latent radiant infrared image $\mathbf{C}$ while preserving sharp scene structures, Stage 1 minimizes the combined energy functional:
\begin{equation}
\label{eq:stage1_objective}
\min_{\mathbf{C}, \mathbf{S}} \frac{1}{2} \|\mathbf{D} - (\mathbf{C} + \mathbf{S})\|_F^2 + \lambda_1 \|\mathbf{S}\|_{S_{1/2}}^{1/2} + \lambda_2 \|\nabla \mathbf{C}\|_1 + \lambda_3 \left\| \frac{\partial^2 \mathbf{C}}{\partial x^2} \right\|_1,
\end{equation}
where:
\begin{itemize}
    \item $\frac{1}{2} \|\mathbf{D} - (\mathbf{C} + \mathbf{S})\|_F^2$ enforces data fidelity with respect to the physical degradation model $\mathbf{D} = \mathbf{C} + \mathbf{S} + \mathbf{N}$ (Eq.~5 and Fig.~6).
    \item $\|\mathbf{S}\|_{S_{1/2}}^{1/2} = \sum_{i=1}^{\min(M, N)} \sqrt{\sigma_i(\mathbf{S})}$ denotes the non-convex Schatten $1/2$-quasi-norm regularizer, which promotes an ultra-sparse singular value spectrum for the column-correlated stripe matrix ($\mathrm{rank}(\mathbf{S}) \le 2$).
    \item $\|\nabla \mathbf{C}\|_1 = \|\nabla_x \mathbf{C}\|_1 + \|\nabla_y \mathbf{C}\|_1$ imposes first-order Total Variation (TV) edge sparsity on the latent thermal scene.
    \item $\left\| \frac{\partial^2 \mathbf{C}}{\partial x^2} \right\|_1$ enforces second-order horizontal directional smoothness, heavily penalizing the cross-column high-frequency ripples induced by vertical stripes while strictly preserving genuine vertical object edges.
\end{itemize}

\textit{Remark on Sign Convention}: In the published manuscript text, Eq.~(11) prints a minor typographical $+C$ within the data fidelity term. To preserve consistency with the physical degradation model $\mathbf{D} = \mathbf{C} + \mathbf{S} + \mathbf{N}$ (Eqs.~5 and 7, and the pipeline diagram Fig.~6 where $\mathbf{C} = \mathbf{D} - \mathbf{S}$), our implementation adheres to the mathematically grounded residual $\mathbf{D} - \mathbf{C} - \mathbf{S}$.

\subsubsection{Variable Splitting and Augmented Lagrangian}
To decouple the non-smooth $L_1$ and non-convex Schatten $1/2$-norms from the linear differential operators, we introduce auxiliary variables:
\begin{equation}
\label{eq:variable_splitting}
\mathbf{G} = \nabla \mathbf{C} = [\mathbf{G}_x, \mathbf{G}_y], \quad \mathbf{H} = \frac{\partial^2 \mathbf{C}}{\partial x^2}.
\end{equation}
The corresponding Augmented Lagrangian function is expressed as:
\begin{align}
\label{eq:augmented_lagrangian}
\mathcal{L}_\rho(\mathbf{C}, \mathbf{S}, \mathbf{G}, \mathbf{H}, \mathbf{\Lambda}_G, \mathbf{\Lambda}_H) &= \frac{1}{2} \|\mathbf{D} - \mathbf{C} - \mathbf{S}\|_F^2 + \lambda_1 \|\mathbf{S}\|_{S_{1/2}}^{1/2} \nonumber \\
&+ \lambda_2 \|\mathbf{G}\|_1 + \lambda_3 \|\mathbf{H}\|_1 \nonumber \\
&+ \frac{\rho}{2} \left\| \nabla \mathbf{C} - \mathbf{G} + \frac{\mathbf{\Lambda}_G}{\rho} \right\|_F^2 \nonumber \\
&+ \frac{\rho}{2} \left\| \frac{\partial^2 \mathbf{C}}{\partial x^2} - \mathbf{H} + \frac{\mathbf{\Lambda}_H}{\rho} \right\|_F^2,
\end{align}
where $\mathbf{\Lambda}_G = [\mathbf{\Lambda}_{Gx}, \mathbf{\Lambda}_{Gy}]$ and $\mathbf{\Lambda}_H$ represent the dual Lagrange multiplier matrices, and $\rho > 0$ is the quadratic penalty parameter.

\subsubsection{Alternating Direction Subproblem Solutions}
At iteration $k+1$, ADMM sequentially optimizes each variable subproblem:

\textbf{1. Clean Scene Subproblem ($\mathbf{C}^{k+1}$)}: Fixing all other variables yields a quadratic minimization problem in $\mathbf{C}$:
\begin{equation}
\label{eq:c_subproblem}
\left( \mathbf{I} + \rho (\nabla_x^T \nabla_x + \nabla_y^T \nabla_y) + \rho \nabla_{xx}^T \nabla_{xx} \right) \mathbf{C}^{k+1} = \mathbf{RHS}^k,
\end{equation}
where $\mathbf{RHS}^k = (\mathbf{D} - \mathbf{S}^k) + \rho \nabla_x^T (\mathbf{G}_x^k - \mathbf{\Lambda}_{Gx}^k/\rho) + \rho \nabla_y^T (\mathbf{G}_y^k - \mathbf{\Lambda}_{Gy}^k/\rho) + \rho \nabla_{xx}^T (\mathbf{H}^k - \mathbf{\Lambda}_H^k/\rho)$.
Under periodic boundary conditions, the differential operators are Block Circulant with Circulant Blocks (BCCC) and are diagonalized via the 2-D Fast Fourier Transform ($\mathcal{F}$):
\begin{equation}
\label{eq:c_fft_solution}
\mathbf{C}^{k+1} = \operatorname{Re}\left( \mathcal{F}^{-1} \left( \frac{\mathcal{F}(\mathbf{RHS}^k)}{1 + \rho (|\mathcal{F}(\nabla_x)|^2 + |\mathcal{F}(\nabla_y)|^2) + \rho |\mathcal{F}(\nabla_{xx})|^2} \right) \right).
\end{equation}

\textbf{2. Stripe Subproblem ($\mathbf{S}^{k+1}$)}: Given the residual $\mathbf{Y}^{k+1} = \mathbf{D} - \mathbf{C}^{k+1}$, the stripe update solves:
\begin{equation}
\label{eq:s_subproblem}
\min_{\mathbf{S}} \frac{1}{2} \|\mathbf{Y}^{k+1} - \mathbf{S}\|_F^2 + \lambda_1 \|\mathbf{S}\|_{S_{1/2}}^{1/2}.
\end{equation}
Let $\mathbf{Y}^{k+1} = \mathbf{U} \mathbf{\Sigma} \mathbf{V}^T$ be the SVD of $\mathbf{Y}^{k+1}$. The optimal singular values $\widehat{\mathbf{\Sigma}}$ are obtained via the closed-form half-thresholding operator $\mathcal{H}_{1/2}(\sigma_i, \tau)$ ($\tau = \lambda_1/\rho$):
\begin{equation}
\mathcal{H}_{1/2}(y, \tau) = \begin{cases}
0, & |y| \le \frac{3}{4} (2\tau)^{2/3} \\
\frac{2}{3} y \left(1 + \cos\left(\frac{2\pi}{3} - \frac{2}{3}\phi\right)\right), & |y| > \frac{3}{4} (2\tau)^{2/3}
\end{cases}
\end{equation}
where $\phi = \arccos\left(\frac{\tau}{8}\left(\frac{|y|}{3}\right)^{-1.5}\right)$. To enforce the physical column-parallel readout topology of the IRFPA, the resulting low-rank matrix is projected onto its column-mean profile:
\begin{equation}
\mathbf{S}^{k+1} = 0.95 \cdot \mathbf{1}_M \left(\frac{1}{M} \mathbf{1}_M^T \mathbf{S}_{\mathrm{svd}}\right) + 0.05 \cdot \mathbf{S}_{\mathrm{svd}}.
\end{equation}

\textbf{3. Auxiliary Gradient Updates ($\mathbf{G}^{k+1}, \mathbf{H}^{k+1}$)}: Decoupled into component-wise ISTA soft-thresholding:
\begin{align}
\mathbf{G}_x^{k+1} &= \operatorname{shrink}\left(\nabla_x \mathbf{C}^{k+1} + \frac{\mathbf{\Lambda}_{Gx}^k}{\rho}, \frac{\lambda_2}{\rho}\right), \label{eq:gx_update} \\
\mathbf{G}_y^{k+1} &= \operatorname{shrink}\left(\nabla_y \mathbf{C}^{k+1} + \frac{\mathbf{\Lambda}_{Gy}^k}{\rho}, \frac{0.2 \lambda_2}{\rho}\right), \label{eq:gy_update} \\
\mathbf{H}^{k+1} &= \operatorname{shrink}\left(\nabla_{xx} \mathbf{C}^{k+1} + \frac{\mathbf{\Lambda}_H^k}{\rho}, \frac{\lambda_3}{\rho}\right). \label{eq:h_update}
\end{align}

\textbf{4. Multiplier Updates}: Dual variables are updated via gradient ascent:
\begin{align}
\mathbf{\Lambda}_{Gx}^{k+1} &= \mathbf{\Lambda}_{Gx}^k + \rho (\nabla_x \mathbf{C}^{k+1} - \mathbf{G}_x^{k+1}), \\
\mathbf{\Lambda}_{Gy}^{k+1} &= \mathbf{\Lambda}_{Gy}^k + \rho (\nabla_y \mathbf{C}^{k+1} - \mathbf{G}_y^{k+1}), \\
\mathbf{\Lambda}_H^{k+1} &= \mathbf{\Lambda}_H^k + \rho (\nabla_{xx} \mathbf{C}^{k+1} - \mathbf{H}^{k+1}).
\end{align}

\textbf{5. Stopping Criterion}: Iterations terminate when the relative change in the restored thermal image falls below tolerance $\epsilon$:
\begin{equation}
\label{eq:stopping_criterion}
\frac{\|\mathbf{C}^{k+1} - \mathbf{C}^k\|_F}{\|\mathbf{C}^k\|_F + 10^{-10}} \le \epsilon \quad \text{or} \quad k \ge K_{\max}.
\end{equation}
